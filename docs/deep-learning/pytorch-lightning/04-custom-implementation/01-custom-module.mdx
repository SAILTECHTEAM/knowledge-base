# Custom LightningModule

This document covers building a custom `LightningModule` — the core abstraction that wraps your model. It includes the full class template, lifecycle hooks, loss patterns, and hyperparameters.

## Full Class Template

```python
import lightning as L
import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.data import DataLoader


class MyLightningModule(L.LightningModule):
    """Custom LightningModule for your model.
    
    Args:
        lr: Learning rate for optimizer.
        hidden_dim: Hidden dimension size.
        dropout: Dropout probability.
    """
    
    def __init__(
        self,
        lr: float = 1e-3,
        hidden_dim: int = 256,
        dropout: float = 0.0,
        num_classes: int = 10,
    ):
        super().__init__()
        # Save hyperparameters to checkpoint
        self.save_hyperparameters()
        
        # Build model layers
        self.encoder = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim * 2),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(hidden_dim * 2, hidden_dim),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(hidden_dim, num_classes),
        )
        
        # Optional: custom attributes
        self.criterion = nn.CrossEntropyLoss()
    
    def forward(self, x):
        """Forward pass.
        
        Args:
            x: Input tensor of shape (batch_size, *).
            
        Returns:
            Output logits of shape (batch_size, num_classes).
        """
        return self.encoder(x)
    
    def training_step(self, batch, batch_idx):
        """Training step - processes one training batch.
        
        Args:
            batch: Tuple of (inputs, targets).
            batch_idx: Index of this batch.
            
        Returns:
            Loss dictionary or scalar loss.
        """
        x, y = batch
        logits = self(x)
        
        loss = self.criterion(logits, y)
        
        # Log metrics
        self.log(
            "train_loss",
            loss,
            on_step=True,
            on_epoch=True,
            prog_bar=True,
            logger=True,
        )
        
        # Calculate accuracy
        preds = torch.argmax(logits, dim=1)
        acc = (preds == y).float().mean()
        self.log(
            "train_acc",
            acc,
            on_step=True,
            on_epoch=True,
            prog_bar=False,
            logger=True,
        )
        
        return loss
    
    def validation_step(self, batch, batch_idx):
        """Validation step - processes one validation batch.
        
        Args:
            batch: Tuple of (inputs, targets).
            batch_idx: Index of this batch.
        """
        x, y = batch
        logits = self(x)
        
        loss = self.criterion(logits, y)
        
        self.log(
            "val_loss",
            loss,
            on_step=False,
            on_epoch=True,
            prog_bar=True,
            logger=True,
        )
        
        # Calculate accuracy
        preds = torch.argmax(logits, dim=1)
        acc = (preds == y).float().mean()
        self.log(
            "val_acc",
            acc,
            on_step=False,
            on_epoch=True,
            prog_bar=True,
            logger=True,
        )
        
        return loss
    
    def test_step(self, batch, batch_idx):
        """Test step - processes one test batch.
        
        Similar to validation_step for most cases.
        """
        x, y = batch
        logits = self(x)
        
        loss = self.criterion(logits, y)
        
        self.log(
            "test_loss",
            loss,
            on_step=False,
            on_epoch=True,
            prog_bar=True,
            logger=True,
        )
        
        preds = torch.argmax(logits, dim=1)
        acc = (preds == y).float().mean()
        self.log(
            "test_acc",
            acc,
            on_step=False,
            on_epoch=True,
            prog_bar=True,
            logger=True,
        )
        
        return loss
    
    def predict_step(self, batch, batch_idx):
        """Predict step - processes one batch for inference.
        
        Args:
            batch: Input batch.
            batch_idx: Index of this batch.
            
        Returns:
            Model predictions.
        """
        x, _ = batch
        logits = self(x)
        return torch.argmax(logits, dim=1)
    
    def configure_optimizers(self):
        """Configure optimizers and learning rate schedulers.
        
        Returns:
            Optimizer, or dict with optimizer and lr_scheduler.
        """
        optimizer = torch.optim.AdamW(
            self.parameters(),
            lr=self.hparams.lr,
            weight_decay=1e-4,
        )
        
        scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(
            optimizer,
            T_max=10,
            eta_min=1e-6,
        )
        
        return {
            "optimizer": optimizer,
            "lr_scheduler": {
                "scheduler": scheduler,
                "monitor": "val_loss",
                "interval": "epoch",
                "frequency": 1,
            },
        }
```

## The `__init__` Method

The `__init__` method accepts configuration parameters and initializes layers:

### Accepting Configuration

```python
def __init__(self, lr: float = 1e-3, hidden_dim: int = 256):
    super().__init__()
    self.save_hyperparameters()
    
    self.layer = nn.Linear(hidden_dim, 10)
```

### Using Explicit kwargs

```python
def __init__(self, **kwargs):
    super().__init__()
    self.save_hyperparameters(kwargs)
    
    hidden_dim = self.hparams.hidden_dim
    self.layer = nn.Linear(hidden_dim, 10)
```

Using `save_hyperparameters()` persists these values in the checkpoint for reproducibility.

## Forward Pass

The `forward()` method defines the model's computation:

```python
def forward(self, x):
    x = self.encoder(x)
    return x
```

Unlike `training_step`, `forward` does not compute loss — it returns the raw model output.

:::tip[forward vs training_step]
- `forward()`: Used during inference (predict, test) and internally by step methods
- `training_step()`: Computes loss and logs metrics
:::

## Step Hooks

Lightning provides step hooks for each phase:

| Hook | Phase | Description |
| :--- | :--- | :--- |
| `training_step` | Train | Process one training batch |
| `training_step_end` | Train | Aggregate batch-level results (optional) |
| `training_epoch_end` | Train | Aggregate epoch-level results (optional) |
| `validation_step` | Validation | Process one validation batch |
| `validation_step_end` | Validation | Aggregate batch-level results (optional) |
| `validation_epoch_end` | Validation | Aggregate epoch-level results (optional) |
| `test_step` | Test | Process one test batch |
| `predict_step` | Inference | Process one batch for prediction |

### training_step

```python
def training_step(self, batch, batch_idx):
    x, y = batch
    logits = self(x)
    
    loss = F.cross_entropy(logits, y)
    
    self.log("train_loss", loss, on_step=True, on_epoch=True)
    
    return loss
```

### validation_step

```python
def validation_step(self, batch, batch_idx):
    x, y = batch
    logits = self(x)
    
    loss = F.cross_entropy(logits, y)
    
    self.log("val_loss", loss, on_step=False, on_epoch=True)
    
    return loss
```

### Multiple Returns

Step methods can return a dictionary for additional outputs:

```python
def training_step(self, batch, batch_idx):
    x, y = batch
    logits = self(x)
    
    loss = F.cross_entropy(logits, y)
    preds = torch.argmax(logits, dim=1)
    
    return {
        "loss": loss,
        "preds": preds,
        "targets": y,
    }
```

## Loss Calculation Patterns

### Cross-Entropy Loss

```python
def training_step(self, batch, batch_idx):
    x, y = batch
    logits = self(x)
    
    loss = F.cross_entropy(logits, y)
    return loss
```

### Custom Loss

```python
def __init__(self, ...):
    super().__init__()
    self.save_hyperparameters()
    self.criterion = nn.MSELoss()

def training_step(self, batch, batch_idx):
    x, y = batch
    logits = self(x)
    
    loss = self.criterion(logits, y)
    return loss
```

### Multi-Task Loss

```python
def training_step(self, batch, batch_idx):
    x, y_cls, y_reg = batch
    pred_cls, pred_reg = self(x)
    
    loss_cls = F.cross_entropy(pred_cls, y_cls)
    loss_reg = F.mse_loss(pred_reg, y_reg)
    
    # Weighted combination
    loss = loss_cls + 0.1 * loss_reg
    
    self.log("train_loss", loss)
    self.log("train_loss_cls", loss_cls)
    self.log("train_loss_reg", loss_reg)
    
    return loss
```

## Logging Patterns

The `self.log()` method logs metrics:

```python
self.log(
    "metric_name",  # Metric name (for logger)
    value,         # Metric value
    on_step=True,  # Log at every step
    on_epoch=True, # Aggregate and log at epoch end
    prog_bar=True,  # Show in progress bar
    logger=True,   # Send to logger
)
```

### Logging Options

| Parameter | When Logged | Use Case |
| :--- | :--- | :--- |
| `on_step=True` | Every batch | Loss (can be noisy) |
| `on_epoch=True` | End of epoch | Summarized metrics |
| `prog_bar=True` | Progress display | Key metrics to watch |
| `logger=True` | External logger | W&B, TensorBoard, etc. |

### Best Practice

```python
# Loss - log both ways for flexibility
self.log("train_loss", loss, on_step=True, on_epoch=True, prog_bar=True)

# Accuracy - log at epoch only
self.log("train_acc", acc, on_step=False, on_epoch=True, prog_bar=True)
```

## Saving and Loading Hyperparameters

The `save_hyperparameters()` method saves init parameters to checkpoint:

```python
def __init__(self, lr: float = 1e-3, hidden_dim: int = 256):
    super().__init__()
    self.save_hyperparameters()
    
    self.layer = nn.Linear(hidden_dim, 10)
```

### Loading

```python
# Automatically restores hyperparameters
model = MyLightningModule.load_from_checkpoint("path/to/ckpt.ckpt")

# Access loaded hyperparameters
print(model.hparams.lr)
print(model.hparams.hidden_dim)
```

### Explicit Save

Save specific parameters:

```python
def __init__(self, lr: float = 1e-3, hidden_dim: int = 256):
    super().__init__()
    self.save_hyperparameters("lr", "hidden_dim")
```

:::caution[Parameter Logging]
Do not log from `__init__`. Use step hooks for logging — the trainer is not yet available in `__init__`.
:::

## Example: Complete MNIST Classifier

```python
import lightning as L
import torch
import torch.nn as nn
import torch.nn.functional as F


class MNISTClassifier(L.LightningModule):
    def __init__(self, lr: float = 1e-3):
        super().__init__()
        self.save_hyperparameters()
        
        self.conv1 = nn.Conv2d(1, 32, 3, padding=1)
        self.conv2 = nn.Conv2d(32, 64, 3, padding=1)
        self.fc1 = nn.Linear(64 * 7 * 7, 128)
        self.fc2 = nn.Linear(128, 10)
        
        self.dropout = nn.Dropout(0.25)
        self.lr = lr
    
    def forward(self, x):
        x = F.relu(F.max_pool2d(self.conv1(x), 2))
        x = F.relu(F.max_pool2d(self.conv2(x), 2))
        x = x.view(x.size(0), -1)
        x = self.dropout(x)
        x = F.relu(self.fc1(x))
        x = self.dropout(x)
        x = self.fc2(x)
        return x
    
    def training_step(self, batch, batch_idx):
        x, y = batch
        logits = self(x)
        loss = F.cross_entropy(logits, y)
        
        self.log("train_loss", loss, on_step=True, on_epoch=True, prog_bar=True)
        return loss
    
    def validation_step(self, batch, batch_idx):
        x, y = batch
        logits = self(x)
        loss = F.cross_entropy(logits, y)
        
        preds = torch.argmax(logits, dim=1)
        acc = (preds == y).float().mean()
        
        self.log("val_loss", loss, on_step=False, on_epoch=True, prog_bar=True)
        self.log("val_acc", acc, on_step=False, on_epoch=True, prog_bar=True)
    
    def configure_optimizers(self):
        return torch.optim.Adam(self.parameters(), lr=self.lr)
```
